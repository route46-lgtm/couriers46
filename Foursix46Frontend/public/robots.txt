# Route46 Couriers — robots.txt
#
# Everything is open to every crawler: traditional search engines, AI search
# and answer engines, and AI training crawlers. The only closed paths are
# admin, the payment flow and the thank-you pages, which carry no public value
# and must never appear in results.
#
# NOTE ON THE AI GROUP BELOW: a crawler obeys ONLY the most specific group that
# matches its name — it does not inherit rules from "User-agent: *". So the
# Disallow lines are repeated in the AI group on purpose. Removing them would
# let those bots crawl /admin and /pay.
# ─────────────────────────────────────────────────────────────────────────────

# ── All crawlers ─────────────────────────────────────────────────────────────
User-agent: *
Allow: /

# Admin and private routes
Disallow: /admin/
Disallow: /admin/login
Disallow: /admin/dashboard

# Payment flow — transactional, no SEO value
Disallow: /pay
Disallow: /pay/success
Disallow: /pay/cancel

# Thank-you pages — no SEO value, avoid duplicate signals
Disallow: /driver-thank-you
Disallow: /shipper-thank-you
Disallow: /contact-thank-you
Disallow: /booking-thank-you
Disallow: /get-a-quote/thank-you
